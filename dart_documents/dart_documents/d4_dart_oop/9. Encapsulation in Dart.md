Using **`_`** underscore before a method/function or property/variable makes it private or encapsulated. So, that it cannot be accessed from outside the class.

We can use **`get`** method or a normal return method (manual getter) for showing private property outside the class. Also, getter (**`get`**) method doesn't take any parameter.

We can use **`set`** method (or helper methods) for setting private or encapsulated property outside the class.

The following lines will give error because we cannot access private members from outside the class cause its encapsulated.

**`throw Exception()`** throws error if pin invalid.

```text
bfc._prepareItem("Chicken Fry"); // won’t work
bfc._shopping("Chicken Fry");
```

```dart

class Restaurant {
  int _id = 187;
  
  int get getId => _id; // get method
  set setId (int value) => _id = value; // set method
  
  void order (String item) {
    print("$item ordered");
    _prepareItem(item);
    _shopping(item);
  }
  
  void _prepareItem (String item) {
    print("$item is being prepared");
  }
  
  void _shopping(String item) {
    print("Shopping for $item");
  }
}

class BankAccount {
  BankAccount({this._pin});
  
  double _balance = 0.0;
  final int _pin;

  double getBalance(int pin) { // helper getter method
    if (pin == _pin) {
      return _balance;
    } else {
      throw Exception("Invalid PIN. Access denied.");
    }
  }
  
  // helper setter method
  set setBalance(double amount) => _balance = amount;
  
  void deposit(double amount) {
    _balance += amount;
    print("Deposited: \$${amount.toStringAsFixed(2)}");
  }
  
  void withdraw(double amount) {
    if (amount > 0 && amount <= _balance) {
      _balance -= amount;
      print("Withdrew: \$${amount.toStringAsFixed(2)}");
    } else if (amount <= 0) {
      print("Withdrawal amount must be positive and greater than 0.");
    } else {
      print("Insufficient funds for withdrawal of \$${amount.toStringAsFixed(2)}");
    }
  }
}

void main () {
  Restaurant bfc = Restaurant();
  bfc.order("Chicken Fry");
  print(bfc.getId);
  bfc.setId = 200;
  print(bfc.getId);

  BankAccount account = BankAccount(pin: 123);
  print(account.getBalance(123));
  account.deposit(500);
  print(account.getBalance(123));
  account.withdraw(200);
  print(account.getBalance(123));
  account.setBalance = 1000;
  print(account.getBalance(123));
}
```
